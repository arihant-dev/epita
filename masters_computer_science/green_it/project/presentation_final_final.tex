%% ============ ACTION 1: STRATEGIC THEORETICAL ANALYSIS ============

\begin{frame}{Action 1.1: Strategic Theoretical Analysis}
\scriptsize
\begin{columns}[T,totalwidth=\textwidth]
\column{0.49\textwidth}
\textbf{Implementation steps (condensed)}
\begin{enumerate}
  \item Audit lifecycle risk in current software portfolio.
  \item Adopt a measurable framework (SCI, Blue Angel, ISO 14044).
  \item Redesign for modularity and backward compatibility.
  \item Commit to long support windows (target: 7+ years).
  \item Remove repair blockers (locks, pairing, contractual limits).
\end{enumerate}

\textbf{Cost-benefit (short vs long term)}
\begin{tabular}{p{1.9cm}p{2.6cm}}
\toprule
Cost now & Benefit later \\
\midrule
Support effort & Lower device-replacement CapEx \\
Refactor cost & Lower maintenance/rebuild cost \\
\bottomrule
\end{tabular}

\column{0.49\textwidth}
\textbf{SWOT (action-specific)}
\begin{itemize}
  \item \textbf{Strength}: regulatory readiness, brand credibility
  \item \textbf{Weakness}: legacy refactor and retraining effort
  \item \textbf{Opportunity}: EU right-to-repair pressure and demand shift
  \item \textbf{Threat}: greenwashing competitors, chipset EOL dependence
\end{itemize}

\textbf{Impact on IT performance}
\begin{itemize}
  \item \textbf{Energy}: lean software lowers CPU/RAM use.
  \item \textbf{Security}: longer patch windows reduce EOL exposure.
  \item \textbf{Resilience}: modular systems are easier to update and audit.
\end{itemize}
\end{columns}
\vspace{0.1cm}
\tiny Sources: \href{https://greensoftware.foundation}{Green Software Foundation}, \href{https://www.blauer-engel.de/en}{Blue Angel}, \href{https://www.europarl.europa.eu/news/en/press-room/20240419IPR20590/right-to-repair-rules-adopted}{EU Right-to-Repair (Parliament)}.
\end{frame}

\end{itemize}

%% ============ ACTION 2: STRATEGIC THEORETICAL ANALYSIS ============

\begin{frame}{Action 2.1: Strategic Theoretical Analysis}
\scriptsize
\begin{columns}[T,totalwidth=\textwidth]
\column{0.49\textwidth}
\textbf{Implementation steps (condensed)}
\begin{enumerate}
  \item Audit current device procurement and replacement cycles.
  \item Establish refurbishment grading criteria (e.g., ISO/IEC 15408, EPEAT).
  \item Partner with certified refurbishers (R2v3, e-Stewards standards).
  \item Update procurement policy to mandate second-hand-first for eligible categories.
  \item Train IT staff to assess, deploy, and support refurbished equipment.
\end{enumerate}

\textbf{Cost-benefit (short vs long term)}
\begin{tabular}{p{1.9cm}p{2.6cm}}
\toprule
Cost now & Benefit later \\
\midrule
Vetting \& sourcing effort & 30--70\% lower acquisition cost \\
Potential shorter warranty & Avoided manufacturing GHG emissions \\
\bottomrule
\end{tabular}

\column{0.49\textwidth}
\textbf{SWOT (action-specific)}
\begin{itemize}
  \item \textbf{Strength}: immediate cost savings; lower embodied carbon
  \item \textbf{Weakness}: perception of lower reliability; limited bulk supply
  \item \textbf{Opportunity}: growing certified refurb market; EU Ecodesign regulation
  \item \textbf{Threat}: planned obsolescence limiting parts availability; greenwashing by resellers
\end{itemize}

\textbf{Impact on IT performance}
\begin{itemize}
  \item \textbf{Energy}: modern refurb units often match new device efficiency.
  \item \textbf{Security}: certified refurbishers wipe and re-image to known-good state.
  \item \textbf{Resilience}: spare-parts availability extends asset lifetime further.
\end{itemize}
\end{columns}
\vspace{0.1cm}
\tiny Sources: \href{https://www.epeat.net}{EPEAT}, \href{https://sustainableelectronics.org/r2-standard/}{R2v3 Standard}, \href{https://commission.europa.eu/law/law-topic/consumer-protection-law/directive-repair-goods_en}{EU Right-to-Repair}.
\end{frame}

  \item Verify that end-of-life take-back is included, not just resale.
\end{itemize}

%% ============ ACTION 3: STRATEGIC THEORETICAL ANALYSIS ============

\begin{frame}{Action 3.1: Strategic Theoretical Analysis}
\scriptsize
\begin{columns}[T,totalwidth=\textwidth]
\column{0.49\textwidth}
\textbf{Implementation steps (condensed)}
\begin{enumerate}
  \item Map all IT assets and identify common failure points.
  \item Create a repair-first policy before replacing laptops, phones, screens, and peripherals.
  \item Keep spare parts for high-volume models (batteries, SSDs, screens, keyboards).
  \item Train IT support teams and publish repair decision rules.
  \item Measure lifetime extension, avoided purchases, and e-waste avoided.
\end{enumerate}

\textbf{Cost-benefit (short vs long term)}
\begin{tabular}{p{2.1cm}p{2.4cm}}
\toprule
Cost now & Benefit later \\
\midrule
Tools, parts stock & Lower replacement CapEx \\
Repair training & Faster recovery \& less downtime \\
\bottomrule
\end{tabular}

\column{0.49\textwidth}
\textbf{SWOT (action-specific)}
\begin{itemize}
  \item \textbf{Strength}: extends device life and reduces procurement pressure
  \item \textbf{Weakness}: repair skills, diagnostics, and parts logistics needed
  \item \textbf{Opportunity}: EU right-to-repair rules improve parts access
  \item \textbf{Threat}: glued designs, parts pairing, and warranty restrictions
\end{itemize}

\textbf{Impact on IT performance}
\begin{itemize}
  \item \textbf{Energy}: repaired devices avoid high manufacturing energy.
  \item \textbf{Security}: repaired assets can be re-imaged and patched before redeployment.
  \item \textbf{Resilience}: spare parts reduce dependency on new supply chains.
\end{itemize}
\end{columns}
\vspace{0.1cm}
\tiny Sources: \href{https://commission.europa.eu/law/law-topic/consumer-protection-law/directive-repair-goods_en}{EU Right-to-Repair}, \href{https://www.ifixit.com/Right-to-Repair}{iFixit Right to Repair}, \href{https://ewastemonitor.info/the-global-e-waste-monitor-2024/}{Global E-waste Monitor 2024}.
\end{frame}

\vspace{0.15cm}
\textbf{How to test any repairability claim}
\begin{itemize}
  \item Check whether manuals, parts, tools, and diagnostics are available to users.
  \item Verify whether repair works without software pairing or vendor authorization.
  \item Look for independent repair scores, not only company marketing.
\end{itemize}
