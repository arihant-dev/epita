%% ============ ACTION 1: STRATEGIC THEORETICAL ANALYSIS ============

\begin{frame}{Slide 1: Strategic Theoretical Analysis}
\scriptsize
\begin{columns}[T,totalwidth=\textwidth]
\column{0.49\textwidth}
\textbf{Implementation steps (condensed)}
\begin{enumerate}
  \item Audit lifecycle risk in current software portfolio.
  \item Adopt a measurable framework (SCI, Blue Angel, ISO 14044).
  \item Redesign for modularity and backward compatibility.
  \item Commit to long support windows (target: 7+ years).
  \item Remove repair blockers (locks, pairing, contractual limits).
\end{enumerate}

\textbf{Cost-benefit (short vs long term)}
\begin{tabular}{p{1.9cm}p{2.6cm}}
\toprule
Cost now & Benefit later \\
\midrule
Support effort & Lower device-replacement CapEx \\
Refactor cost & Lower maintenance/rebuild cost \\
\bottomrule
\end{tabular}

\column{0.49\textwidth}
\textbf{SWOT (action-specific)}
\begin{itemize}
  \item \textbf{Strength}: regulatory readiness, brand credibility
  \item \textbf{Weakness}: legacy refactor and retraining effort
  \item \textbf{Opportunity}: EU right-to-repair pressure and demand shift
  \item \textbf{Threat}: greenwashing competitors, chipset EOL dependence
\end{itemize}

\textbf{Impact on IT performance}
\begin{itemize}
  \item \textbf{Energy}: lean software lowers CPU/RAM use.
  \item \textbf{Security}: longer patch windows reduce EOL exposure.
  \item \textbf{Resilience}: modular systems are easier to update and audit.
\end{itemize}
\end{columns}

\end{frame}

\begin{frame}{Slide 5: Greenwashing Case Study (Apple)}
\scriptsize
\begin{tabular}{p{2.5cm}p{5.6cm}p{2.2cm}}
\toprule
Claim & Evidence snapshot & Verdict \\
\midrule
``100\% renewable operations'' & Valid for all corporate operations (stores, offices, data centers in 44 countries), but manufacturing/Scope 3 remains dominant. & Partial \\
``Carbon-neutral products'' & Offset-heavy claims are legally and methodologically contested in multiple jurisdictions. & Contested \\
``USB-C is eco-forward'' & Standardization helps future waste, but legacy Lightning accessories became stranded rapidly. & Mixed / contested \\
``Updates extend lifespan'' & Support exists for newer models, but HOP complaint on parts pairing/repair barriers triggered a Paris prosecutor-led investigation. & Inconsistent \\

\end{tabular}

\vspace{0.15cm}
\textbf{How to test any sustainability claim}
\begin{itemize}
  \item Ask if lifecycle emissions (Scope 3) are included.
  \item Distinguish actual reductions from offsets.
  \item Check right-to-repair behavior and third-party certification.
\end{itemize}

\begin{frame}{Slide 1: Strategic Theoretical Analysis}
\scriptsize
\begin{columns}[T,totalwidth=\textwidth]
\column{0.49\textwidth}
\textbf{Implementation steps (condensed)}
\begin{enumerate}
  \item Audit current device procurement and replacement cycles.
  \item Establish refurbishment grading criteria (e.g., ISO/IEC 15408, EPEAT).
  \item Partner with certified refurbishers (R2v3, e-Stewards standards).
  \item Update procurement policy to mandate second-hand-first for eligible categories.
  \item Train IT staff to assess, deploy, and support refurbished equipment.
\end{enumerate}

\textbf{Cost-benefit (short vs long term)}
\begin{tabular}{p{1.9cm}p{2.6cm}}
\toprule
Cost now & Benefit later \\
\midrule
Vetting \& sourcing effort & 30--70\% lower acquisition cost \\
Potential shorter warranty & Avoided manufacturing GHG emissions \\
\bottomrule
\end{tabular}

\column{0.49\textwidth}
\textbf{SWOT (action-specific)}
\begin{itemize}
  \item \textbf{Strength}: immediate cost savings; lower embodied carbon
  \item \textbf{Weakness}: perception of lower reliability; limited bulk supply
  \item \textbf{Opportunity}: growing certified refurb market; EU Ecodesign regulation
  \item \textbf{Threat}: planned obsolescence limiting parts availability; greenwashing by resellers
\end{itemize}

\textbf{Impact on IT performance}
\begin{itemize}
  \item \textbf{Energy}: modern refurb units often match new device efficiency.
  \item \textbf{Security}: certified refurbishers wipe and re-image to known-good state.
  \item \textbf{Resilience}: spare-parts availability extends asset lifetime further.
\end{itemize}
\end{columns}
\vspace{0.1cm}
\tiny Sources: \href{https://www.epeat.net}{EPEAT}, \href{https://sustainableelectronics.org/r2-standard/}{R2v3 Standard}, \href{https://commission.europa.eu/law/law-topic/consumer-protection-law/directive-repair-goods_en}{EU Right-to-Repair}.
\end{frame}

%% ============ ACTION 3: GREENWASHING ============

\begin{frame}{Slide 5: Greenwashing Case Study (Amazon Renewed)}
\scriptsize
\begin{tabular}{p{2.5cm}p{5.6cm}p{2.2cm}}
\toprule
Claim & Evidence snapshot & Verdict \\
\midrule
``Certified refurbished'' label & No single external standard; sellers self-certify under Amazon's internal checklist. No independent audit required. & Misleading \\
``Like new'' condition guarantee & Consumer complaints document functional defects, missing accessories, and incorrect grading at significant rates. & Inconsistent \\
``Sustainable choice'' badge & Badge applied algorithmically based on Amazon's own criteria; no lifecycle emissions data published per product. & Unverified \\
``Supports circular economy'' & Amazon's core business model drives new-product sales; Renewed is a secondary channel with no explicit take-back or end-of-life commitment. & Partial / contested \\
\bottomrule
\end{tabular}

\vspace{0.15cm}
\textbf{How to test any second-hand/refurb claim}
\begin{itemize}
  \item Check for third-party certification (R2v3, e-Stewards, PAS 2060).
  \item Ask whether a lifecycle carbon assessment (LCA) exists for the refurb process.
  \item Verify that end-of-life take-back is included, not just resale.
\end{itemize}
