\section{Objective and Scope}

\subsection{Final lab objective}
Implement and deploy a 3-service mobility platform on AWS with:
\begin{itemize}
    \item private runtime network for services,
    \item controlled outbound traffic through NAT,
    \item IAM baseline for deploy/runtime identities,
    \item WAF protection at CloudFront edge,
    \item API entry through API Gateway to private services.
\end{itemize}

\subsection{Implemented repositories and components}
\begin{itemize}
    \item \texttt{lab\_final/repos/urbanmove-auth}
    \item \texttt{lab\_final/repos/urbanmove-ingestion-routing}
    \item \texttt{lab\_final/repos/urbanmove-api}
    \item \texttt{lab\_final/frontend/static} (multi-page UI: home, login, operations, control)
    \item \texttt{lab\_final/infra/aws-cli} (infra provisioning)
    \item \texttt{lab\_final/infra/deploy} (service deployment and smoke test)
\end{itemize}

\section{Architecture and Traffic Flow}

\subsection{Architecture diagram}
\labimage{../architecture/eraser/architecture_design.png}{Final architecture exported from Eraser}

\subsection{Request path}
\begin{enumerate}
    \item User requests UI/API via CloudFront.
    \item CloudFront routes:
    \begin{itemize}
        \item default path to S3 frontend bucket,
        \item \texttt{/api/*} path to API Gateway.
    \end{itemize}
    \item API Gateway forwards private integration traffic through VPC Link to internal NLB.
    \item Internal NLB forwards to API service in private subnet.
    \item API service calls Auth and Ingestion/Routing over gRPC.
\end{enumerate}

\subsection{Outbound external feed path}
Traffic to the government transport API is initiated by private services and exits via NAT Gateway and Internet Gateway.

\subsection{Backup and restore drill (non-destructive)}
\Needspace{16\baselineskip}
\begin{lstlisting}[language=bash]
# SSH to ingest host
cd lab_final/infra/deploy
source ./common.sh
ssh "${SSH_OPTS[@]}" "${SSH_USER}@${INGEST_HOST}"

# Backup production database
sudo docker exec urbanmove-postgres pg_dump -U urbanmove -d urbanmove > /tmp/urbanmove_backup.sql

# Restore into separate database
sudo docker exec -i urbanmove-postgres psql -U urbanmove -d postgres -c "DROP DATABASE IF EXISTS urbanmove_restore;"
sudo docker exec -i urbanmove-postgres psql -U urbanmove -d postgres -c "CREATE DATABASE urbanmove_restore;"
cat /tmp/urbanmove_backup.sql | sudo docker exec -i urbanmove-postgres psql -U urbanmove -d urbanmove_restore

# Verify restored schema
sudo docker exec -i urbanmove-postgres psql -U urbanmove -d urbanmove_restore -c "\dt"
\end{lstlisting}
