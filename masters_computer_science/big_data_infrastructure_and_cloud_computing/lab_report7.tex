\section{GitLab + Lambda}

\subsection{LAB 07 required steps and execution}
LAB 07 defines this flow for GitLab + Lambda: create project from template, create IAM deployment user, attach required permissions, add AWS credentials to CI/CD variables, run pipeline, read endpoint, test endpoint.

\textbf{Exact identifiers used in our run:}
\begin{itemize}
    \item GitLab project: \texttt{iarihanti/aws-lambda-demo}
    \item IAM user: \texttt{gitlab\_demo\_lambda}
    \item IAM policies: \texttt{AWSCloudFormationFullAccess} and \texttt{AWSLambda\_FullAccess}
    \item Region: \texttt{us-east-1}
    \item Deployed endpoint: \url{https://lcqn6muh65.execute-api.us-east-1.amazonaws.com/production/hello}
\end{itemize}

\textbf{Execution steps:}
\begin{enumerate}
    \item Created the GitLab serverless project template.
    \item Created IAM deployment user and attached Lambda/CloudFormation permissions.
    \item Added \texttt{AWS\_ACCESS\_KEY\_ID} and \texttt{AWS\_SECRET\_ACCESS\_KEY} in GitLab CI/CD variables.
    \item Triggered pipeline and verified all jobs finished in success state.
    \item Opened generated API endpoint and validated response with Postman.
\end{enumerate}

\subsection{Exact .gitlab-ci.yml used (GitLab + Lambda)}
\begin{lstlisting}
image: node:18

variables:
  AWS_REGION: us-east-1
  AWS_DEFAULT_REGION: $AWS_REGION

stages:
  - test
  - deploy_function
  - test_deployed_function
  - deploy_pages

test:
  stage: test
  script:
    - npm install
    - npm test

production:
  stage: deploy_function
  before_script:
    - FAILURE_MESSAGE="Must define $AWS_ACCESS_KEY_ID and $AWS_SECRET_ACCESS_KEY. Add keys to $CI_PROJECT_URL/-/settings/ci_cd"
    - test -z "$AWS_SECRET_ACCESS_KEY" && echo $FAILURE_MESSAGE && exit 1
    - test -z "$AWS_ACCESS_KEY_ID" && echo $FAILURE_MESSAGE && exit 1
    - export AWS_DEFAULT_REGION="$AWS_REGION"
  script:
    - npm install
    - npm run deploy -- --stage production --verbose
  environment: production
  only:
     - master
  artifacts:
    paths:
      - stack.json
    expire_in: 2 weeks

postdeploy_test:
  stage: test_deployed_function
  only:
    - master
  script:
    - npm install
    - STACK_JSON_FILE=./stack.json npm test featureTests

pages:
  stage: deploy_pages
  script:
    - cp stack.json ./public/stack.json
  environment: production_pages
  only:
    - master
  artifacts:
    paths:
      - ./public/
\end{lstlisting}

\subsection{Exact .gitlab-ci.yml used (GitLab + ECS/Fargate)}
\begin{lstlisting}
stages:
  - test
  - build
  - deploy

variables:
  APP_PORT: "8080"
  AWS_PAGER: ""
  DOCKER_HOST: "tcp://docker:2375"
  DOCKER_TLS_CERTDIR: ""
  DOCKER_DRIVER: overlay2

test:
  stage: test
  image: golang:1.22-alpine
  script:
    - go test ./...

build-image:
  stage: build
  image: docker:24.0.7
  services:
    - name: docker:24.0.7-dind
      command: ["--tls=false"]
  needs:
    - test
  rules:
    - if: '$CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH'
  before_script:
    - apk add --no-cache aws-cli
  script:
    - ': "${AWS_ACCOUNT_ID:?AWS_ACCOUNT_ID is required}"'
    - ': "${AWS_REGION:?AWS_REGION is required}"'
    - export ECR_REGISTRY="${AWS_ACCOUNT_ID}.dkr.ecr.${AWS_REGION}.amazonaws.com"
    - export IMAGE_REPOSITORY="${ECR_REPOSITORY:-$CI_PROJECT_PATH_SLUG}"
    - export IMAGE_URI="${ECR_REGISTRY}/${IMAGE_REPOSITORY}"
    - |
      if ! aws ecr describe-repositories --region "$AWS_REGION" --repository-names "$IMAGE_REPOSITORY" >/dev/null 2>&1; then
        echo "Creating missing ECR repository: ${IMAGE_REPOSITORY}"
        aws ecr create-repository --region "$AWS_REGION" --repository-name "$IMAGE_REPOSITORY" >/dev/null
      fi
    - aws ecr get-login-password --region "$AWS_REGION" | docker login --username AWS --password-stdin "$ECR_REGISTRY"
    - docker build --pull -t "${IMAGE_URI}:${CI_COMMIT_SHORT_SHA}" .
    - docker push "${IMAGE_URI}:${CI_COMMIT_SHORT_SHA}"
    - |
      if [ "${CI_COMMIT_BRANCH}" = "${CI_DEFAULT_BRANCH}" ]; then
        docker tag "${IMAGE_URI}:${CI_COMMIT_SHORT_SHA}" "${IMAGE_URI}:latest"
        docker push "${IMAGE_URI}:latest"
      fi

deploy-ecs:
  stage: deploy
  image: alpine:3.20
  before_script:
    - apk add --no-cache aws-cli
  needs:
    - build-image
  environment:
    name: ecs-fargate
  rules:
    - if: '$CI_COMMIT_BRANCH == $CI_DEFAULT_BRANCH'
  script:
    - ': "${AWS_ACCOUNT_ID:?AWS_ACCOUNT_ID is required}"'
    - ': "${AWS_REGION:?AWS_REGION is required}"'
    - ': "${ECS_EXECUTION_ROLE_ARN:?ECS_EXECUTION_ROLE_ARN is required}"'
    - export ECR_REGISTRY="${AWS_ACCOUNT_ID}.dkr.ecr.${AWS_REGION}.amazonaws.com"
    - export IMAGE_REPOSITORY="${ECR_REPOSITORY:-$CI_PROJECT_PATH_SLUG}"
    - export IMAGE_URI="${ECR_REGISTRY}/${IMAGE_REPOSITORY}:${CI_COMMIT_SHORT_SHA}"
    - export TASK_FAMILY="${ECS_TASK_FAMILY:-$CI_PROJECT_PATH_SLUG}"
    - export CONTAINER_NAME="${ECS_CONTAINER_NAME:-app}"
    - export CONTAINER_PORT="${APP_PORT:-8080}"
    - export CPU="${ECS_CPU:-256}"
    - export MEMORY="${ECS_MEMORY:-512}"
    - export ECS_CLUSTER_NAME="${ECS_CLUSTER:-${CI_PROJECT_NAME:-$CI_PROJECT_PATH_SLUG}}"
    - export ECS_SERVICE_NAME="${ECS_SERVICE:-${CI_PROJECT_NAME:-$CI_PROJECT_PATH_SLUG}}"
    - |
      case "$CONTAINER_PORT" in
        ''|*[!0-9]*)
          echo "APP_PORT must be numeric, got '$CONTAINER_PORT'"
          exit 1
          ;;
      esac
    - |
      CLUSTER_STATUS="$(aws ecs describe-clusters --clusters "$ECS_CLUSTER_NAME" --query 'clusters[0].status' --output text 2>/dev/null || true)"
      if [ -z "$CLUSTER_STATUS" ] || [ "$CLUSTER_STATUS" = "None" ]; then
        echo "ECS cluster '${ECS_CLUSTER_NAME}' not found. Set ECS_CLUSTER to an existing cluster."
        exit 1
      fi
      SERVICE_STATUS="$(aws ecs describe-services --cluster "$ECS_CLUSTER_NAME" --services "$ECS_SERVICE_NAME" --query 'services[0].status' --output text 2>/dev/null || true)"
      if [ -z "$SERVICE_STATUS" ] || [ "$SERVICE_STATUS" = "None" ]; then
        echo "ECS service '${ECS_SERVICE_NAME}' not found in cluster '${ECS_CLUSTER_NAME}'. Set ECS_SERVICE to an existing service."
        exit 1
      fi
    - echo "Deploying ${IMAGE_URI} to ECS service ${ECS_SERVICE_NAME} in cluster ${ECS_CLUSTER_NAME}"
    - sed -e "s|__TASK_FAMILY__|${TASK_FAMILY}|g" -e "s|__EXECUTION_ROLE_ARN__|${ECS_EXECUTION_ROLE_ARN}|g" -e "s|__IMAGE_URI__|${IMAGE_URI}|g" -e "s|__CONTAINER_NAME__|${CONTAINER_NAME}|g" -e "s|__CONTAINER_PORT__|${CONTAINER_PORT}|g" -e "s|__CPU__|${CPU}|g" -e "s|__MEMORY__|${MEMORY}|g" deploy/ecs-task-definition.json.tpl > rendered-task-definition.json
    - TASK_DEF_ARN="$(aws ecs register-task-definition --cli-input-json file://rendered-task-definition.json --query 'taskDefinition.taskDefinitionArn' --output text)"
    - aws ecs update-service --cluster "$ECS_CLUSTER_NAME" --service "$ECS_SERVICE_NAME" --task-definition "$TASK_DEF_ARN" --force-new-deployment
    - aws ecs wait services-stable --cluster "$ECS_CLUSTER_NAME" --services "$ECS_SERVICE_NAME"
\end{lstlisting}
